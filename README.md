        function handleDriveImageFallback(imgElement, fileId) {
            const currentAttempt = parseInt(imgElement.getAttribute('data-attempt') || '0', 10);
            if (currentAttempt === 0) {
                imgElement.setAttribute('data-attempt', '1');
                imgElement.src = 'https://drive.google.com/thumbnail?id=' + fileId + '&sz=w1000';

                imgElement.setAttribute('data-attempt', '2');
                imgElement.src = 'https://drive.google.com/uc?export=view&id=' + fileId;
            } else if (currentAttempt === 2) {
                imgElement.setAttribute('data-attempt', '3');
                imgElement.src = 'https://lh3.googleusercontent.com/d/' + fileId;
            }
        }
